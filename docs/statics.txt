esp-rtos
    scheduler.rs
        pub(crate) struct Scheduler {
            // a bit bigger than to store it in cell
            inner: Mutex<RawMutex, RefCell<SchedulerState>>,
        }
        then 
        impl Scheduler {
            pub(crate) fn with<R>(&self, cb: impl FnOnce(&mut SchedulerState) -> R) -> R {
                self.with_shared(|shared| cb(&mut *unwrap!(shared.try_borrow_mut())))
            }

            pub(crate) fn with_shared<R>(&self, cb: impl FnOnce(&RefCell<SchedulerState>) -> R) -> R {
                self.inner.lock(|shared| cb(shared))
            }
            ...
        }

embassy-executor
    raw/run_queue_critical_section.rs
        use critical_section::{CriticalSection, Mutex};
        // use the original/general critical_section's Mutex, but critical_section
        // impl is using the multicore aware esp_sync::RawMutex;
        // but there is some difference using CriticalSectionRawMutex or esp_sync::RawMutex
        // the former will spinlock on the same (static CRITICAL_SECTION:) memory for every mutex
        // tha latter can be spinlock on a specific static
        pub(crate) struct RunQueueItem {
            // Cell is enough, since we only replace its contents
            next: Mutex<Cell<Option<TaskRef>>>,
        }

        use critical_section::{CriticalSection, Mutex};
        pub(crate) struct RunQueue {
            head: Mutex<Cell<Option<TaskRef>>>,
        }

    raw/state_critical_section.rs
        use critical_section::{CriticalSection, Mutex};
