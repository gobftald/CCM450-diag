// Time driver
// trait for establishing API for a "plugable" implementation
pub trait Driver: Send + Sync + 'static {
    /// Return the current timestamp in ticks.
    ///
    /// Implementations MUST ensure that:
    /// - This is guaranteed to be monotonic, i.e. a call to now() will always return
    ///   a greater or equal value than earler calls. Time can't "roll backwards".
    /// - It "never" overflows.
    fn now(&self) -> u64;

    /// Sets the callback function and 'ctx' to be called when the alarm triggers.
    /// The callback may be called from any context (interrupt or thread mode).
    ///
    /// 'ctx' is -- in our case -- a *mut () to an SyncExecutor
    /// so callback function can act as a particular SyncExecutor,
    /// in our case its simply use it pender and send a signal to the top Executor
    fn set_alarm_callback(&self, alarm: AlarmHandle, callback: fn(*mut ()), ctx: *mut ());

    /// Sets an alarm at the given timestamp. When the current timestamp reaches the alarm
    /// timestamp, the provided callback function will be called.
    fn set_alarm(&self, alarm: AlarmHandle, timestamp: u64) -> bool;
}

// Links to an even non existing ("plugable") implementation.

// providing a macro (see later) and these links eliminates this problem.

    fn _embassy_time_set_alarm(alarm: AlarmHandle, timestamp: u64) -> bool;
}

impl Driver for EmbassyTimer {
    fn now(&self) -> u64 {
        EmbassyTimer::now()     // calls device related part - esp-hal/src/embassy/time_driver_timg.rs
    }

    unsafe fn allocate_alarm(&self) -> Option<AlarmHandle> {
        critical_section::with(|cs| {
            // walk through alarms
            for (i, alarm) in self.alarms.borrow(cs).iter().enumerate() {
                // look for the first non allocated
                if !alarm.allocated.get() {
                    // set alarm so it is not overwritten
                    alarm.allocated.set(true);
                    // this function is currrently empty, see below
                    self.on_alarm_allocated(i);
                    // return with the handle of allocated alarm, which is
                    // its AlarmHandle wrapped index in the 'alarms' array
                    // AlarmHandle comes from embassy-time-driver
                    return Some(AlarmHandle::new(i as u8));
                }
            }
            None
        })
    }

    fn set_alarm_callback(
        &self,
        alarm: AlarmHandle,
        callback: fn(*mut ()),
        ctx: *mut (),
    ) {
        // get alarm's indec from AlarmHandle
        let n = alarm.id() as usize;
        critical_section::with(|cs| {
            // get alarm by index
            let alarm = &self.alarms.borrow(cs)[n];
            // save callback function
            alarm.callback.set(Some((callback, ctx)));
        })
    }

    fn set_alarm(&self, alarm: embassy_time_driver::AlarmHandle, timestamp: u64) -> bool {
        // calls EmbassyTimer::set_alarm(&self, alarm, timestamp)
        self.set_alarm(alarm, timestamp)
    }

use core::task::Waker;

/// Timer queue
pub trait TimerQueue {
    /// Schedules a waker in the queue to be awoken at moment `at`.
    /// If this moment is in the past, the waker might be awoken immediately.
    fn schedule_wake(&'static self, at: u64, waker: &Waker);
}

extern "Rust" {
    fn _embassy_time_schedule_wake(at: u64, waker: &Waker);
}

/// Schedule the given waker to be woken at `at`.
pub fn schedule_wake(at: u64, waker: &Waker) {
    unsafe { _embassy_time_schedule_wake(at, waker) }
}

/// Set the TimerQueue implementation.
///
/// See the module documentation for an example.
#[macro_export]
macro_rules! timer_queue_impl {
    (static $name:ident: $t: ty = $val:expr) => {
        static $name: $t = $val;

        #[no_mangle]
        fn _embassy_time_schedule_wake(at: u64, waker: &core::task::Waker) {
            <$t as $crate::TimerQueue>::schedule_wake(&$name, at, waker);
        }
    };
}

fn _embassy_time_schedule_wake(at: u64, waker: &core::task::Waker) {
    // Casting TimerQueue trait to TimerQueue struct, then the linkable function
    // calls the TimerQueue struct's implementation with 'fabricated' method calls,
    <TimerQueue as ::embassy_time_queue_driver::TimerQueue>::schedule_wake(
        &TIMER_QUEUE,
        at,
        waker,
    );
}

#[cfg(feature = "integrated-timers")]
impl embassy_time_queue_driver::TimerQueue for TimerQueue {
    // get task from waker, then set task's expires if arg 'at' is less then current one
    fn schedule_wake(&'static self, at: u64, waker: &core::task::Waker) {
        let task = waker::task_from_waker(waker);   // embassy-executor/raw/waker.rs
        let task = task.header();
        unsafe {
            let expires_at = task.expires_at.get();
            task.expires_at.set(expires_at.min(at));
        }
    }
}

// Timer struct is a future that completes at a specified Instant.
// Ticker struct is an ssynchronous stream that yields every Duration (at uniform intervals), even if blocking work is
// performed between ticks

// We only implement Timer's after function?

use core::future::Future;
use core::pin::Pin;
use core::task::{Context, Poll};

use crate::{Duration, Instant};

pub struct Timer {
    expires_at: Instant,
    yielded_once: bool,
}

impl Timer {
    pub fn after(duration: Duration) -> Self {
        // creates a new Timer with now + duration
        Self {
            expires_at: Instant::now() + duration,
            // the first run will be used for 'enqueuing a timer'
            // e.g. calling embassy_time_queue_driver::schedule_wake, see below
            yielded_once: false,
        }
    }
}

// implementing our own future
impl Future for Timer {
    type Output = ();
    fn poll(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
        // if expires return with Poll::Ready(())
        if self.yielded_once && self.expires_at <= Instant::now() {
            Poll::Ready(())
        } else {
            // in the first run call schedule_wake to set task's expire_at
            embassy_time_queue_driver::schedule_wake(self.expires_at.as_ticks(), cx.waker());
            self.yielded_once = true;
            Poll::Pending
        }
    }
}
