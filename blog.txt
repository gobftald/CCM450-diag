openocd -c 'set ESP_RTOS none' -f board/esp32c3-builtin.cfg
riscv32-esp-elf-gdb esp-minimal
    
    target extended-remote :3333        // connect to openocd
    mon reset halt                      // monitor (command to openocd) reset then halt cpu
    flushregs                           // all registers are 0, except pc, which is 0x40000000
                                        // after reset and halt, pc is in its default address
    b *0x42010020                       // address of loaded program (= start address in elf file)
    c                                   // continue

    set mem inaccessible-by-default off
    set remote hardware-watchpoint-limit 8  // for esp32c3

    Ctrl-x a        // enter,re-enter or leave TUI (character window) mode
    Ctrl-x 1        // use a TUI layout with only one window
    Ctrl-x 2        // use a TUI layout with at least two windows
